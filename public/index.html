Integração Bidirecional Google Sheets

✓ Sincronização em tempo real JSONP + Banco Firebase ativa

O aplicativo lê a planilha do Google Docs sem bloqueio de CORS (JSONP) a cada 10 segundos e sincroniza com o banco de dados Firebase Firestore.

Para permitir que o salvamento no app edite diretamente as células da planilha no Google Docs sem exigir login Google individual em cada celular: abra a planilha → Extensões > Apps Script → cole o código abaixo → clique em Implantar > Nova implantação > App da Web (Acesso: Qualquer pessoa) e cole a URL gerada acima. 

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheetId = data.sheet_id || '1Vrv-BaXEsZtSsB6G3Ix-eK3Xk4rcos-HYp9zQJVVcM0';
  var ss = SpreadsheetApp.openById(sheetId);
  var sheet = ss.getSheets()[0];
  var values = sheet.getDataRange().getValues();
  var target = String(data.leito || '').trim().toUpperCase();
  var headerRow = values[1] || [];
  var hasTs = String(headerRow[3] || '').trim().toUpperCase().indexOf('T.S') === 0;
  for (var i = 2; i < values.length; i++) {
    if (String(values[i][0] || '').trim().toUpperCase() === target) {
      if (hasTs) {
        sheet.getRange(i + 1, 2, 1, 6).setValues([[
          data.nome || '',
          data.paridade || '',
          data.tipo_sanguineo || '',
          data.alto_risco || '',
          data.internacao_por || '',
          data.observacoes || ''
        ]]);
      } else {
        sheet.getRange(i + 1, 2, 1, 5).setValues([[
          data.nome || '',
          data.paridade || '',
          data.alto_risco || '',
          data.internacao_por || '',
          data.observacoes || ''
        ]]);
      }
      return ContentService.createTextOutput(JSON.stringify({ok: true}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ok: false}))
    .setMimeType(ContentService.MimeType.JSON);
}
